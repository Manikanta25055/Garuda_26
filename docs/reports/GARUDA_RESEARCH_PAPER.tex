\section{Research Problem}

\textbf{Problem definition:}
Existing smart surveillance systems rely heavily on cloud infrastructure, introducing
three critical failure modes: (1) \emph{privacy risk} — raw video streams leave the
premises and are stored on third-party servers; (2) \emph{latency} — round-trip to the
cloud adds 200–2{,}000 ms, making real-time response impossible; and (3)
\emph{connectivity dependency} — any network outage disables the system entirely.
Existing edge deployments either rely on general-purpose CPUs (insufficient throughput
for multi-model pipelines) or proprietary hardware locked to vendor cloud services.

\textbf{Research objective:}
To design and evaluate a privacy-preserving, fully edge-based intelligent surveillance
system that performs all AI inference locally on commodity embedded hardware, achieves
real-time detection throughput ($\geq$30 FPS), and is resilient to both physical bypass
attempts (weapon concealment) and digital bypass attempts (camera spoofing).

\textbf{Novel contributions (stated here for reference throughout the paper):}
\begin{itemize}[noitemsep]
  \item Fully local three-model cascade on a single 13-TOPS NPU — zero cloud dependency
  \item MiDaS monocular depth variance as a physically grounded anti-spoofing signal
  \item Context-aware decision engine that fuses time-of-day, depth, and classification
        evidence before raising any alert
  \item Integrated multi-modal interface: web dashboard, voice assistant (Narada), and
        email alert with OTP authentication
\end{itemize}

\section{Technical Architecture}

\subsection*{System Components}

\begin{description}[noitemsep]
  \item[Camera input] IMX708 12 MP sensor via \texttt{libcamerasrc}, running at 52 FPS
  \item[AI detection module] YOLOv8s Scout — detects knife, hammer, scissors, person
  \item[Threat classifier] MobileNetV2 Inspector — binary Safe/Weapon classifier on
        cropped person regions
  \item[Anti-spoof module] MiDaS Small Reality Checker — monocular depth estimator
  \item[Alert engine] Email (SMTP + OTP), voice TTS (Narada), danger\_sightings log
  \item[Authentication module] bcrypt-hashed passwords, OTP gate, session JWT, login
        lockout after 5 failed attempts
  \item[Database / storage] JSON logs in \texttt{system\_logs/}, 7-day rolling retention
        with daily backups; SQLite for user records
  \item[UI] FastAPI web dashboard (live MJPEG stream + controls); voice command
        interface via SpeechRecognition
\end{description}

All three HEF model files are loaded simultaneously into a single Hailo VDevice
(\texttt{/dev/hailo0}), which multiplexes inference across the Hailo-8L DSP fabric.

\subsection*{Data Flow (Algorithmic)}

\begin{enumerate}[noitemsep]
  \item \textbf{Capture} — \texttt{libcamerasrc} pulls frame from IMX708; pushed into
        \texttt{frame\_queue} on CPU Core 0.
  \item \textbf{Preprocessing} — Core 1 dequeues frame; resizes to $640\times640$,
        normalises to $[0,1]$ float32 for YOLOv8s input.
  \item \textbf{Object detection} — YOLOv8s Scout runs on Hailo DSP; outputs bounding
        boxes with class label and confidence score.
  \item \textbf{Classification (weapon vs.\ person)} — If label is \emph{weapon} (knife,
        hammer, scissors): escalate directly. If label is \emph{person}: crop ROI,
        resize to $224\times224$, pass to MobileNetV2 Inspector.
  \item \textbf{Anti-spoofing / depth check} — Full frame passed to MiDaS Small;
        depth variance $\sigma^2$ computed over the scene. If $\sigma^2 < 0.05$: flag
        as \texttt{Spoof\_Attempt}.
  \item \textbf{Decision logic} — Context-aware engine on Core 2 fuses outputs from
        steps 4 and 5 with time-of-day and mode flags (DND, IDLE, NIGHT).
  \item \textbf{Alert generation} — If alert condition met: email dispatched (SMTP,
        30 min cooldown), voice TTS announcement, danger\_sightings.txt updated.
  \item \textbf{Logging} — Every event written to \texttt{perm\_detection\_log.txt} and
        the rolling JSON session log.
\end{enumerate}

\section{Mathematical / Algorithmic Modelling}

\textbf{(a) Detection confidence (Recall):}
\begin{equation}
  P_{\text{detect}} = \frac{TP}{TP + FN}
  \label{eq:recall}
\end{equation}

\textbf{(b) False Alarm Rate:}
\begin{equation}
  FAR = \frac{FP}{FP + TN}
  \label{eq:far}
\end{equation}

\textbf{(c) System latency decomposition:}
\begin{equation}
  T_{\text{total}} = T_{\text{capture}} + T_{\text{processing}} + T_{\text{alert}}
  \label{eq:latency}
\end{equation}
where $T_{\text{capture}} \approx 2$~ms (libcamerasrc + queue),
$T_{\text{processing}} \approx 18.38$~ms (Hailo HW inference),
and $T_{\text{alert}} \approx 1.6$~ms (postprocess + decision). Total: $T_{\text{total}} \approx 22$~ms for single-model path; $\approx 27$~ms for full cascade.

\textbf{(d) Anti-spoofing decision rule:}
\begin{equation}
  \text{label} =
  \begin{cases}
    \texttt{Spoof\_Attempt} & \text{if } \sigma^2_{\text{depth}} < 0.05 \\
    \texttt{RealPerson}     & \text{otherwise}
  \end{cases}
  \label{eq:spoof}
\end{equation}
Flat screens and printed photographs produce $\sigma^2 < 0.005$; real persons in indoor scenes produce $\sigma^2 \in [0.022, 0.10]$.

\section{Security Analysis}

\textbf{Threat model:}
\begin{itemize}[noitemsep]
  \item \emph{Unauthorized physical access} — intruder enters monitored space
  \item \emph{Camera spoofing} — attacker presents a photograph or screen to bypass
        person detection
  \item \emph{Credential attacks} — brute-force or credential-stuffing the web dashboard
  \item \emph{Network interception} — MITM on local LAN to intercept alert emails or
        API calls
\end{itemize}

\textbf{Defense mechanisms:}
\begin{itemize}[noitemsep]
  \item OTP email authentication gate — a 6-digit OTP is required to arm/disarm the
        system; codes expire after 5 minutes
  \item bcrypt password hashing (cost factor 12) for all stored credentials
  \item Rate limiting on the FastAPI login endpoint (5 attempts per minute)
  \item Login lockout — account locked for 15 minutes after 5 consecutive failures
  \item JWT session tokens with 1-hour expiry and server-side rotation on each refresh
  \item MiDaS depth anti-spoofing (Section III, Eq.~\ref{eq:spoof}) — eliminates
        flat-screen bypass; $\sigma^2 < 0.05$ rejects 100\% of tested screen attacks
  \item All video processing is on-device; no frame leaves the LAN
\end{itemize}

\section{Intelligence Layer (Novelty)}

The system moves beyond simple detection $\to$ alert by implementing a
\textbf{context-aware decision engine}. The decision matrix:

\begin{table}[h]
\centering
\caption{Context-aware decision matrix}
\begin{tabular}{@{}llll@{}}
\toprule
Detection     & Time     & Depth $\sigma^2$ & Action \\
\midrule
Weapon        & Any      & Any              & HIGH alert (immediate) \\
Person        & Night    & $\geq 0.05$      & HIGH alert + voice \\
Person        & Day      & $\geq 0.05$      & LOG only (normal) \\
Person        & Any      & $< 0.05$         & Flag SPOOF, silent log \\
No detection  & Night    & Any              & Presence window check \\
\bottomrule
\end{tabular}
\end{table}

Additional context inputs: DND mode (suppresses voice), IDLE mode (suppresses email),
NIGHT mode (lowers confidence threshold to 0.35 for reduced-light sensitivity),
and a 30-minute email cooldown to prevent alert fatigue.

\section{Literature Review}

\textbf{Edge AI surveillance systems.}
Redmon et al.\ \cite{redmon2016} introduced YOLO as the first single-stage detector
suitable for real-time embedded deployment. Ultralytics YOLOv8 \cite{ultralytics2023}
extended this with anchor-free heads and improved small-object recall. Adarsh et al.\
\cite{adarsh2020} benchmarked YOLO variants on edge hardware, concluding that
hardware-accelerated NPUs are necessary to sustain $>$30 FPS on devices below 10 W.

\textbf{Privacy-preserving AI.}
Dwork et al.\ \cite{dwork2006} formalised differential privacy as a mathematical
framework for data release without individual leakage. McMahan et al.\ \cite{mcmahan2017}
demonstrated federated learning as an alternative to centralised training; Garuda
applies both philosophies operationally — no training data or inference result leaves
the device.

\textbf{Smart home security.}
Singh et al.\ \cite{singh2021} surveyed IoT security systems, identifying latency and
privacy as the two unsolved constraints. Zuboff \cite{zuboff2019} analysed the systemic
privacy risks of always-on cloud surveillance. Agrawal et al.\ \cite{agrawal2021}
proposed on-device threat classification using MobileNet, achieving 94\% accuracy at
29 FPS on a Coral TPU — comparable to our MobileNetV2 Inspector (99.6\%, Hailo-8L).

\textbf{Positioning:}
Unlike all surveyed systems, Garuda performs complete inference locally without any
cloud dependency, combines three complementary models on a single low-cost NPU, and
adds a physically grounded anti-spoofing layer absent from all cited works.

\section{Novel Contributions}

\begin{enumerate}[noitemsep]
  \item \textbf{Fully local AI-based surveillance} — all inference runs on the
        Raspberry Pi 5 + Hailo-8L; no frame, no model output, and no log leaves the
        device network perimeter.
  \item \textbf{Three-model cascade on a single 13-TOPS NPU} — YOLOv8s Scout,
        MobileNetV2 Inspector, and MiDaS Small Reality Checker multiplexed via Hailo
        VDevice; first demonstrated combination of detection, classification, and
        depth estimation on Hailo-8L in a single pipeline.
  \item \textbf{Physically grounded anti-spoofing} — monocular depth variance
        ($\sigma^2 < 0.05$) as a hardware-independent liveness signal, eliminating
        flat-screen bypass with 0\% pass-through in all tested conditions.
  \item \textbf{Context-aware decision engine} — fuses detection class, depth,
        time-of-day, and user-defined mode flags before raising any alert, reducing
        false alerts from 11.2 to 0.8 per hour.
  \item \textbf{Integrated multi-modal interface} — web dashboard (FastAPI + MJPEG),
        voice assistant (Narada, SpeechRecognition TTS), and OTP-gated email alerts;
        all running concurrently on the same Pi 5 without interfering with inference.
\end{enumerate}
